     Phase 4: Explore & Discovery - Implementation Plan               
                                                                      
     Branch Name: feature/phase-4-explore                             
                                                                      
     Overview                                                         
                                                                      
     Phase 4 implements the consultant discovery and browsing         
     feature, allowing users to:                                      
     - Browse verified consultants with filtering (domain, rating,    
      price)                                                          
     - Search consultants by name/keywords                            
     - View detailed consultant profiles                              
     - See consultation/subscription plans and reviews

     1.3 GET /api/consultants Query Parameters                        
                                                                      
     domainId: String (optional) - Filter by domain                   
     subDomainId: String (optional) - Filter by subdomain             
     minRating: double (optional) - Minimum rating (0-5)              
     maxPrice: int (optional) - Maximum price in smallest currency    
      unit                                                            
     search: String (optional) - Search name, headline, bio           
     page: int (default 0) - Page number                              
     pageSize: int (default 20, max 50) - Items per page              
     sortBy: String (rating|price|experience) - Sort field            
     sortOrder: String (asc|desc) - Sort direction                    
                                                                      
     1.4 Response Format                                              
                                                                      
     {                                                                
       "consultants": [                                               
         {                                                            
           "id": "...",                                               
           "userId": "...",                                           
           "user": { "name": "...", "image": "..." },                 
           "headline": "...",                                         
           "rating": 4.8,                                             
           "experience": 5.0,                                         
           "domain": { "id": "...", "name": "..." },                  
           "subDomains": [...],                                       
           "languages": ["English", "Hindi"],                         
           "minPrice": 2000,                                          
           "priceCurrency": "INR",                                    
           "totalMenteesHelped": 50                                   
         }                                                            
       ],                                                             
       "pagination": {                                                
         "page": 0,                                                   
         "pageSize": 20,                                              
         "totalCount": 150,                                           
         "totalPages": 8,                                             
         "hasNextPage": true,                                         
         "hasPreviousPage": false                                     
       }                                                              
     }

     2.2 Consultant Entity                                            
                                                                      
     @freezed                                                         
     class Consultant with _$Consultant {                             
       const factory Consultant({                                     
         required String id,                                          
         required String userId,                                      
         required String name,                                        
         String? image,                                               
         double? experience,                                          
         @Default(0) double rating,                                   
         String? headline,                                            
         String? bio,                                                 
         DomainEntity? domain,                                        
         @Default([]) List<SubDomainEntity> subDomains,               
         @Default([]) List<String> languages,                         
         @Default([]) List<String> toolsAndTechnologies,              
         bool? isVerified,                                            
         @Default(0) int totalMenteesHelped,                          
         int? minPrice,                                               
         String? priceCurrency,                                       
       }) = _Consultant;                                              
     }                                                                
                                                                      
     2.3 ExploreFilters Entity                                        
                                                                      
     @freezed                                                         
     class ExploreFilters with _$ExploreFilters {                     
       const factory ExploreFilters({                                 
         String? domainId,                                            
         String? subDomainId,                                         
         double? minRating,                                           
         int? maxPrice,                                               
         String? searchQuery,                                         
         @Default('rating') String sortBy,                            
         @Default(true) bool sortDesc,                                
       }) = _ExploreFilters;                                          
                                                                      
       bool get hasFilters => domainId != null || subDomainId !=      
     null ||                                                          
                              minRating != null || maxPrice !=        
     null;                                                            
     }
