 Step B3: Database Setup (lib/database/prisma_client.dart)

 import 'package:postgres/postgres.dart' as pg;
 import 'package:prisma_flutter_connector/runtime.dart';

 class DatabaseClient {
   static DatabaseClient? _instance;
   late final SupabaseAdapter _adapter;
   late final QueryExecutor _executor;

   DatabaseClient._();

   static Future<DatabaseClient> initialize(String connectionUrl) async {
     if (_instance != null) return _instance!;

     _instance = DatabaseClient._();
     final uri = Uri.parse(connectionUrl);

     final connection = await pg.Connection.open(
       pg.Endpoint(
         host: uri.host,
         port: uri.port,
         database: uri.pathSegments.first,
         username: uri.userInfo.split(':')[0],
         password: uri.userInfo.split(':')[1],
       ),
       settings: pg.ConnectionSettings(sslMode: pg.SslMode.require),
     );

     _instance!._adapter = SupabaseAdapter(connection);
     _instance!._executor = QueryExecutor(adapter: _instance!._adapter);

     return _instance!;
   }

   QueryExecutor get executor => _executor;
 }

 Step B4: Auth Service (lib/services/auth_service.dart)

 class AuthService {
   final QueryExecutor _executor;
   final JwtService _jwtService;

   // Sign in with email
   Future<Map<String, dynamic>> signInWithEmail(String email, String
 password) async {
     final users = await _executor.executeQueryAsMaps(
       JsonQueryBuilder()
         .model('User')
         .action(QueryAction.findMany)
         .where({'email': email})
         .build()
     );

     if (users.isEmpty) throw AuthException('Invalid credentials');

     final user = users.first;
     // Note: NextAuth stores hashed passwords in Account table, not User
     // Need to check Account table for credentials provider

     final session = await _createSession(user['id']);
     return {'user': user, 'session': session};
   }

   // Sign up with email
   Future<Map<String, dynamic>> signUpWithEmail(String email, String
 password, String? name) async {
     // Check if user exists
     final existing = await _executor.executeQueryAsMaps(
       JsonQueryBuilder()
         .model('User')
         .action(QueryAction.findMany)
         .where({'email': email})
         .build()
     );

     if (existing.isNotEmpty) throw AuthException('Email already 
 registered');

     // Create user
     final hashedPassword = BCrypt.hashpw(password, BCrypt.gensalt());
     final userId = Uuid().v4();

     final user = await _executor.executeQueryAsMaps(
       JsonQueryBuilder()
         .model('User')
         .action(QueryAction.create)
         .data({
           'id': userId,
           'email': email,
           'name': name,
           'role': 'CONSULTEE',
           'createdAt': DateTime.now().toIso8601String(),
           'updatedAt': DateTime.now().toIso8601String(),
         })
         .build()
     );

     // Create account for credentials
     await _executor.executeQueryAsMaps(
       JsonQueryBuilder()
         .model('Account')
         .action(QueryAction.create)
         .data({
           'id': Uuid().v4(),
           'userId': userId,
           'type': 'credentials',
           'provider': 'credentials',
           'providerAccountId': userId,
           'password': hashedPassword,
         })
         .build()
     );

     final session = await _createSession(userId);
     return {'user': user.first, 'session': session};
   }
 }

 Step B5: API Endpoints

 POST /api/auth/sign-in/email (routes/api/auth/sign-in/email.dart)
 import 'package:dart_frog/dart_frog.dart';

 Future<Response> onRequest(RequestContext context) async {
   if (context.request.method != HttpMethod.post) {
     return Response(statusCode: 405);
   }

   final body = await context.request.json() as Map<String, dynamic>;
   final email = body['email'] as String?;
   final password = body['password'] as String?;

   if (email == null || password == null) {
     return Response.json(
       statusCode: 400,
       body: {'error': {'message': 'Email and password required'}},
     );
   }

   try {
     final authService = context.read<AuthService>();
     final result = await authService.signInWithEmail(email, password);
     return Response.json(body: result);
   } catch (e) {
     return Response.json(
       statusCode: 401,
       body: {'error': {'message': e.toString()}},
     );
   }
 }

 GET /api/auth/session (routes/api/auth/session.dart)
 Future<Response> onRequest(RequestContext context) async {
   if (context.request.method != HttpMethod.get) {
     return Response(statusCode: 405);
   }

   final token =
 context.request.headers['authorization']?.replaceFirst('Bearer ', '');
   if (token == null) {
     return Response.json(body: {'session': null, 'user': null});
   }

   try {
     final jwtService = context.read<JwtService>();
     final authService = context.read<AuthService>();

     final payload = jwtService.verify(token);
     final session = await authService.getSession(payload['sessionId']);
     final user = await authService.getUser(session['userId']);

     return Response.json(body: {'session': session, 'user': user});
   } catch (e) {
     return Response.json(body: {'session': null, 'user': null});
   }
 }

 Step B6: CORS Middleware (routes/_middleware.dart)

 import 'package:dart_frog/dart_frog.dart';

 Handler middleware(Handler handler) {
   return (context) async {
     // Handle CORS preflight
     if (context.request.method == HttpMethod.options) {
       return Response(
         headers: {
           'Access-Control-Allow-Origin': '*',
           'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, 
 OPTIONS',
           'Access-Control-Allow-Headers': 'Content-Type, Authorization',
         },
       );
     }

     final response = await handler(context);
     return response.copyWith(
       headers: {
         ...response.headers,
         'Access-Control-Allow-Origin': '*',
       },
     );
   };
 }

 Step B7: Main Entry Point (main.dart)

 import 'dart:io';
 import 'package:dart_frog/dart_frog.dart';
 import 'package:dotenv/dotenv.dart';
 import 'lib/database/prisma_client.dart';
 import 'lib/services/auth_service.dart';
 import 'lib/services/jwt_service.dart';

 Future<HttpServer> run(Handler handler, InternetAddress ip, int port)
 async {
   // Load env
   final env = DotEnv()..load(['.env']);

   // Initialize database
   final db = await DatabaseClient.initialize(env['DATABASE_URL']!);

   // Create services
   final jwtService = JwtService(env['JWT_SECRET']!);
   final authService = AuthService(db.executor, jwtService);

   // Add providers to handler
   final handlerWithProviders = handler
       .use(provider<AuthService>((_) => authService))
       .use(provider<JwtService>((_) => jwtService));

   return serve(handlerWithProviders, ip, port);
 }

 Title: "Migrate from NextAuth.js to Better Auth for cross-platform auth
 consistency"

 Issue Content:

 ## Summary
 The mobile app uses `better_auth_flutter` which expects Better Auth
 backend endpoints. Currently, the web app uses NextAuth.js. This creates
 an architecture where we need a separate Dart Frog backend for mobile.

 ## Why Migrate
 - **Single auth backend** for both web and mobile
 - **Consistent API** across platforms
 - **Simplified architecture** (eliminate Dart Frog auth backend)
 - **Better Auth** has native support for Flutter/Dart clients

 ## Technical Steps
 1. Install `better-auth` package
 2. Create `/api/auth/[...betterauth]/route.ts`
 3. Configure Better Auth with Prisma adapter
 4. Migrate session handling from NextAuth to Better Auth
 5. Update client-side auth hooks
 6. Test all OAuth providers (Google, GitHub)
 7. Remove NextAuth configuration

 ## Database Compatibility
 Better Auth uses its own session/account tables but can be configured to
 use existing Prisma schema with adapters.

 ## Timeline
 To be scheduled after Flutter mobile app MVP is complete.
